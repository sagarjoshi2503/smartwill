    <h3>Who can be a Witness?</h3>
    <p>Under the Indian Succession Act, 1925, any person of sound mind who is a major (18 or older) and capable of signing their own name can act as a witness. At least two witnesses are required, and neither needs to know what the Will actually says.</p>
    <ul>
      <li><strong>Signing:</strong> each witness must see the testator sign the Will, or be told directly that the signature is theirs. Both witnesses then sign in the testator's presence.</li>
      <li><strong>Not a beneficiary:</strong> avoid choosing someone who is also inheriting under the Will, or their spouse — for some communities, being a witness can void that person's own inheritance.</li>
      <li><strong>An executor can witness:</strong> the person managing your estate may also act as a witness.</li>
      <li><strong>Family members can witness</strong>, but an independent, impartial person is generally the safer choice to avoid any future dispute.</li>

  let currentStep = 0;

  function renderTabs(current) {
    const el = document.getElementById('tabsBlock');
    el.innerHTML = TAB_STEPS.map(t => {
      const cls = ['tab'];
      if (t.n === current) cls.push('active');
      else if (t.n < current) cls.push('done');
      return `<div class="${cls.join(' ')}" onclick="goTo(${t.n})">${t.label}</div>`;

  function goTo(step) {
    currentStep = step;
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.querySelectorAll(`.page[data-step="${step}"]`).forEach(p => p.classList.add('active'));
    renderTabs(step);
    window.scrollTo({top:0, behavior:'smooth'});
  }

  function escapeHtml(str) {
    return str.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  function toggleBequeathedOther(selectEl) {
    const otherInput = selectEl.parentElement.querySelector('.bequeathed-other');
    if (!otherInput) return;
    otherInput.style.display = selectEl.value === '__other__' ? 'block' : 'none';
  }

  function refreshBequeathedDropdowns() {
    const names = getBeneficiaryNames();
    document.querySelectorAll('.bequeathed-select').forEach(sel => {
      const prevValue = sel.value;
      const otherInput = sel.parentElement.querySelector('.bequeathed-other');
      const options = ['<option value="">Select recipient...</option>']
        .concat(names.map(n => `<option value="${escapeHtml(n)}">${escapeHtml(n)}</option>`))
        .concat(['<option value="__other__">Other / Not listed above</option>']);
      sel.innerHTML = options.join('');
      if (prevValue && names.includes(prevValue)) {
        sel.value = prevValue;
      } else if (prevValue === '__other__') {
        sel.value = '__other__';
      } else if (prevValue) {
        sel.value = '__other__';
        if (otherInput && !otherInput.value) otherInput.value = prevValue;
      }
      toggleBequeathedOther(sel);
    });
  }

  /* ---------- Will document generator ---------- */
  function fillOrBlank(value, len) {
    if (value && String(value).trim()) return String(value).trim();
    return '_'.repeat(len || 20);
  }

  function selVal(el) {
    if (!el) return '';
    const v = el.value;
    if (!v || v === 'Select' || v === 'Select...') return '';
    return v;
  }

  function idVal(id) {
    const el = document.getElementById(id);
    return el ? el.value.trim() : '';
  }

  function getBequeathedValue(block) {
    const sel = block.querySelector('.bequeathed-select');
    if (!sel) return '';
    if (sel.value === '__other__') {
      const other = block.querySelector('.bequeathed-other');
      return other ? other.value.trim() : '';
    }
    return sel.value;
  }

  function getAssetEntries(containerId) {
    const container = document.getElementById(containerId);
    if (!container) return [];
    return Array.from(container.querySelectorAll('.repeat-block')).map(block => ({
      desc: block.querySelector('.asset-desc') ? block.querySelector('.asset-desc').value.trim() : '',
      bequeathed: getBequeathedValue(block),
      relationship: selVal(block.querySelector('.asset-relationship')),
      idtype: selVal(block.querySelector('.asset-idtype')),
      idnumber: block.querySelector('.asset-idnumber') ? block.querySelector('.asset-idnumber').value.trim() : '',
    })).filter(e => e.desc || e.bequeathed || e.idnumber);
  }

  function formatAssetLine(label, entries) {
    if (!entries.length) {
      return `${label}: ${'_'.repeat(30)} Bequeathed to: ${'_'.repeat(25)}. Relationship: Spouse / Son / Daughter / Other: ______, bearing Aadhaar Card /Pan Card/ Passport/ Voter ID/ Driving License Number: ${'_'.repeat(20)}.`;
    }
    return entries.map(e => {
      const relText = e.relationship ? e.relationship : '______';
      const idText = e.idtype ? `${e.idtype} No.` : 'Aadhaar Card /Pan Card/ Passport/ Voter ID/ Driving License Number';
      return `${label}: ${fillOrBlank(e.desc,30)} Bequeathed to: ${fillOrBlank(e.bequeathed,25)}. Relationship: Spouse / Son / Daughter / Other: ${relText}, bearing ${idText}: ${fillOrBlank(e.idnumber,20)}.`;
    }).join('\n');
  }

  function getWitnessData(index) {
    const blocks = document.querySelectorAll('#witnessList .repeat-block');
    const block = blocks[index];
    if (!block) return null;
    return {
      name: block.querySelector('.witness-name') ? block.querySelector('.witness-name').value.trim() : '',
      age: block.querySelector('.witness-age') ? block.querySelector('.witness-age').value.trim() : '',
      relation: selVal(block.querySelector('.witness-relation')),
      parent: block.querySelector('.witness-parent') ? block.querySelector('.witness-parent').value.trim() : '',
      marital: selVal(block.querySelector('.witness-marital')),
      nationality: block.querySelector('.witness-nationality') ? block.querySelector('.witness-nationality').value.trim() : '',
      occupation: selVal(block.querySelector('.witness-occupation')),

      address: block.querySelector('.witness-address') ? block.querySelector('.witness-address').value.trim() : '',
    };
  }

  function witnessParagraph(w) {
    if (!w) w = {};

  }

  function generateWillText() {
    const name = idVal('testatorName');
    const pan = idVal('testatorPAN');
    const aadhaar = idVal('testatorAadhaar');
    const relation = selVal(document.getElementById('testatorRelation'));
    const parentName = idVal('testatorParentName');
    const age = idVal('testatorAge');
    const maritalStatus = selVal(document.getElementById('maritalStatus'));
    const isMarried = maritalStatus === 'Married';
    const maritalText = maritalStatus ? maritalStatus.toLowerCase() : 'unmarried/married';
    const nationality = idVal('testatorNationality');
    const occupation = selVal(document.getElementById('testatorOccupation'));
    const address = idVal('testatorAddress');
    const spouseName = idVal('spouseName');
    const spouseAadhaar = idVal('spouseAadhaar');

    const w1 = witnessParagraph(getWitnessData(0));
    const w2 = witnessParagraph(getWitnessData(1));

    const footer = "Testator's Signature: ___________________ Witness 1: _________ Witness 2: _________";

    let spouseLine;
    if (isMarried) {
      spouseLine = `I am married to ${fillOrBlank(spouseName,30)}, bearing Aadhar No.${fillOrBlank(spouseAadhaar,15)} and I have (one/two/three etc) son, namely, ${'_'.repeat(30)} and (one/two/three etc) daughter, namely, ${'_'.repeat(30)}`;
    } else {
      spouseLine = `I have (one/two/three etc) son, namely, ${'_'.repeat(30)} and (one/two/three etc) daughter, namely, ${'_'.repeat(30)}`;
    }

    const houseFlat = formatAssetLine('House / Flat', getAssetEntries('houseFlatList'));
    const landPlot = formatAssetLine('Land / Plot', getAssetEntries('landPlotList'));
    const commercial = formatAssetLine('Commercial Property', getAssetEntries('commercialPropertyList'));
    const vehicle = formatAssetLine('Vehicle / Car', getAssetEntries('vehicleList'));
    const jewellery = formatAssetLine('Jewellery & Heirlooms', getAssetEntries('jewelleryList'));
    const digital = formatAssetLine('Social Media / Digital', getAssetEntries('digitalAssetList'));
    const ip = formatAssetLine('Intellectual Property', getAssetEntries('intellectualPropertyList'));

    const residual = getResidualData();
    const residualText = residual
      ? `${fillOrBlank(residual.relationship,15)}, ${fillOrBlank(residual.name,30)}, nationality ${fillOrBlank(residual.nationality,15)}, occupation (retired/in service/business/profession/homemaker) ${fillOrBlank(residual.occupation,20)}, bearing ${residual.idtype ? residual.idtype + ' No.' : 'Aadhaar Card/PAN Card/Passport/Voter ID/Driving Licence Number'}: ${fillOrBlank(residual.idnumber,20)}`
      : `${'_'.repeat(15)} (Relationship), ${'_'.repeat(30)}, nationality ${'_'.repeat(15)}, occupation (retired/in service/business/profession/homemaker) ${'_'.repeat(20)}, bearing Aadhaar Card/PAN Card/Passport/Voter ID/Driving Licence Number: ${'_'.repeat(20)}`;

    const w1Name = (getWitnessData(0) || {}).name;
    const w2Name = (getWitnessData(1) || {}).name;

    return `${footer}

WILL

I, ${fillOrBlank(name,35)}, having PAN ${fillOrBlank(pan,15)}, Aadhar no ${fillOrBlank(aadhaar,15)} ${relation ? relation.toLowerCase() : 'son/daughter of'} ${fillOrBlank(parentName,30)}, aged ${fillOrBlank(age,4)}, ${maritalText}, nationality ${fillOrBlank(nationality,15)}, occupation (retired/in service/business/profession/homemaker) ${fillOrBlank(occupation,20)}, resident of ${fillOrBlank(address,35)}, ${spouseLine} and in the presence of two following witnesses:

a) ${w1}

b) ${w2}

make my last and final WILL.

I am making this last WILL and testament of mine voluntarily and without any compulsion or pressure from any source or person and in sound health and disposing state of mind. I have not been influenced, cajoled or coerced in any manner to write this WILL. I do hereby revoke all my wills, if any, previously made by me.

I own the following movable and immovable properties which are all self-acquired or built out of my own earning and income and have absolute power of disposal of the same.

I bequeath my specific assets to the designated beneficiaries as outlined below:

A. Financial Assets:
I bequeath all my financial assets including Bank Accounts, Fixed Deposits (FDs), Recurring Deposits (RDs), Public Provident Fund (PPF), Life Insurance, Stocks, Mutual Funds, Cryptocurrency (Crypto), Digital Wallets, National Pension System (NPS), Bonds, Alternative Investment Fund (AIF), Specialized Investment Fund (SIF), and Portfolio Management Services (PMS) entirely to the nominees registered in those financial instruments.

B. Immovable Property:
${houseFlat}

${footer}

${landPlot}

${commercial}

C. Motor Vehicles:
${vehicle}

D. Personal & Valuables:
${jewellery}

${footer}

E. Digital & Miscellaneous Assets:
${digital}

${ip}

I hereby declare, direct, and devise that all the Rest and Residue of my estate, including any property or assets, both movable and immovable, which I may acquire after the execution of this Will, or which has been inadvertently omitted from this document, shall be given entirely to ${residualText}.

Signature of Testator/Testatrix______________________________

I have fully understood the contents, significance and implications contained in this WILL which has been executed out of my free will, and choice. There has been no misrepresentation in regard to this WILL, and no one has any right to object and/or to challenge this WILL as this is culmination of my discretion and best for me to safeguard my interest and interest of my family.

Testator/Testatrix understands and approves the contents of document before signing and was not forced to do so by any person.

Signature of Testator/Testatrix:
Name of Testator/Testatrix: ${fillOrBlank(name,35)}
Place: ${'_'.repeat(20)}

WITNESSES

  }
